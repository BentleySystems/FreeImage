cmake_minimum_required(VERSION 3.20)
project(FreeImageEXR LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(IMATH_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(IMATH_BUILD_PYTHON OFF CACHE BOOL "" FORCE)
set(OPENEXR_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(OPENEXR_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(OPENEXR_BUILD_PYTHON OFF CACHE BOOL "" FORCE)
set(OPENEXR_INSTALL_DOCS OFF CACHE BOOL "" FORCE)
set(OPENEXR_FORCE_INTERNAL_DEFLATE ON CACHE BOOL "" FORCE)
set(OPENEXR_FORCE_INTERNAL_OPENJPH ON CACHE BOOL "" FORCE)
set(OPENEXR_FORCE_INTERNAL_ZSTD ON CACHE BOOL "" FORCE)
set(OPENEXR_FORCE_INTERNAL_IMATH ON CACHE BOOL "" FORCE)
add_subdirectory(upstream/imath imath)
add_subdirectory(upstream/openexr openexr)
get_filename_component(FREEIMAGE_EXR_LIBRARY "${CMAKE_BINARY_DIR}/../OpenEXR.lib" ABSOLUTE)
add_custom_command(
    OUTPUT "${FREEIMAGE_EXR_LIBRARY}"
    COMMAND "${CMAKE_AR}" /NOLOGO "/OUT:${FREEIMAGE_EXR_LIBRARY}"
        "$<TARGET_FILE:OpenEXR>" "$<TARGET_FILE:OpenEXRCore>"
        "$<TARGET_FILE:Iex>" "$<TARGET_FILE:IlmThread>" "$<TARGET_FILE:Imath>"
    DEPENDS OpenEXR OpenEXRCore Iex IlmThread Imath
    VERBATIM)
add_custom_target(FreeImageEXR ALL DEPENDS "${FREEIMAGE_EXR_LIBRARY}")
